\section{Basic Python Syntax}

\begin{frame}[fragile]
\frametitle{Comments in Python}
\begin{itemize}
\item Comments are used for descriptive or explanatory purposes
\item Comments are non-executable statements
\item Comments start with \texttt{\#} at the beginning of every line
\end{itemize}

\begin{lstlisting}
# This is a comment
print("Hello World")  # This is also a comment
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Variables}
\begin{itemize}
\item Variables store data values
\item Python has no syntax for declaring variables like C, C++
\item Variables are created when you assign a value
\end{itemize}

\begin{lstlisting}
Number1 = 20
string1 = "Stephen"
print(Number1)
print(string1)
\end{lstlisting}

\begin{lstlisting}
x = "John"
# is the same as
x = 'John'
\end{lstlisting}
\end{frame}

\begin{frame}
\frametitle{Variable Naming Rules}
\begin{enumerate}
\item Must begin with an English alphabet
\item Case-sensitive (SUM ≠ sum)
\item Can be alphanumeric but not numeric-alpha
\item Reserved keywords cannot be variable names
\item Can be of any length
\item Only underscore allowed as special character
\item No need to declare data type
\end{enumerate}
\end{frame}

\begin{frame}[fragile]
\frametitle{Print Statement}
\begin{itemize}
\item Used for printing output
\item For text concatenation, use \texttt{+} character
\item For numbers, \texttt{+} works as mathematical operator
\end{itemize}

\begin{lstlisting}
String2 = "Bye"
print("Good" + String2)
# Output: GoodBye

A = 3
B = 8
print(A + B)
# Output: 11
\end{lstlisting}
\end{frame}

\section{Data Types}

\begin{frame}[fragile]
\frametitle{Numeric Types in Python}
Three numeric types in Python:
\begin{itemize}
\item \textbf{int} - Integer (whole numbers)
\item \textbf{float} - Floating point numbers
\item \textbf{complex} - Complex numbers
\end{itemize}

\begin{lstlisting}
A = 25      # int
B = 14.32   # float
C = 5j      # complex

print(type(A))  # <class 'int'>
print(type(B))  # <class 'float'>
print(type(C))  # <class 'complex'>
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Integer (int)}
\begin{itemize}
\item Whole numbers without fractional part
\item Can be positive or negative
\end{itemize}

\begin{lstlisting}
A = 100
B = 12345
print(type(A))  # <class 'int'>
print(type(B))  # <class 'int'>
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Float}
\begin{itemize}
\item Numbers with fractional or decimal part
\item Can be positive or negative
\item Can use scientific notation with "e"
\end{itemize}

\begin{lstlisting}
Number1 = 15.32
Number2 = 17.0

# Scientific notation
A = 35e3  # 35000.0
B = 12E4  # 120000.0
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Complex Numbers}
\begin{itemize}
\item Numbers with real and imaginary parts
\item Imaginary part denoted with letter "j"
\end{itemize}

\begin{lstlisting}
A = 15+9j
B = -12j
print(type(A))  # <class 'complex'>
print(type(B))  # <class 'complex'>
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Type Casting}
Convert from one data type to another:

\begin{lstlisting}
A = 23      # int
B = 15.4    # float
C = 3+4j    # complex

# Convert from int to float
A1 = float(A)

# Convert from float to int
B1 = int(B)

# Convert from int to complex
C1 = complex(A)

print(A1, type(A1))
print(B1, type(B1))
print(C1, type(C1))
\end{lstlisting}

\textbf{Note:} Cannot convert complex numbers to other types.
\end{frame}

\begin{frame}[fragile]
\frametitle{Random Numbers}
\begin{itemize}
\item No built-in random number generator function
\item Use \texttt{random} module
\end{itemize}

\begin{lstlisting}
import random

# Generate random number between 1 and 20
print(random.randrange(1, 20))
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{String Literals}
\begin{itemize}
\item String constants can be in single or double quotes
\item \texttt{'welcome'} is same as \texttt{"welcome"}
\end{itemize}

\begin{lstlisting}
print("welcome")
print('welcome')

# Assign string to variable
String3 = "UNESWA"
print(String3)

# Multiline strings
Attitude = """My students are
good students in programming,
because they love programming."""
print(Attitude)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Strings as Arrays}
\begin{itemize}
\item Strings are arrays of Unicode characters
\item No separate character data type
\item Use square brackets to access elements
\end{itemize}

\begin{lstlisting}
a = "Welcome to UNESWA"
print(a[1])  # Output: e

# Substring
b = "Welcome to UNESWA"
print(b[3:9])  # Output: come to

# String length
print(len(a))  # Output: 17
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{String Methods}
\begin{lstlisting}
a = "Welcome to UNESWA"

# Convert to lowercase
print(a.lower())  # welcome to uneswa

# Convert to uppercase
print(a.upper())  # WELCOME TO UNESWA

# Remove whitespace
a = " Welcome to UNESWA "
print(a.strip())  # Welcome to UNESWA

# Replace characters
print(a.replace("E", "I"))  # Welcome to UNISWA

# Split string
print(a.split(","))  # ['Welcome to UNESWA']
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Identity Operators}
Compare objects, not values:

\begin{lstlisting}
a = ["Mango", "Orange"]
b = ["Mango", "Orange"]
c = a

print(a is c)     # True
print(a is b)     # False
print(a == b)     # True
\end{lstlisting}

\begin{itemize}
\item \texttt{is} - Returns True if variables are same object
\item \texttt{is not} - Returns True if variables are not same object
\end{itemize}
\end{frame}

\begin{frame}[fragile]
\frametitle{Membership Operators}
Test if sequence is present in object:

\begin{lstlisting}
x = ["apple", "banana"]

print("banana" in x)        # True
print("pineapple" not in x) # True
\end{lstlisting}

\begin{itemize}
\item \texttt{in} - Returns True if sequence is present
\item \texttt{not in} - Returns True if sequence is not present
\end{itemize}
\end{frame}

\section{Data Structures}

\begin{frame}
\frametitle{Python Collections}
Four collection data types:
\begin{itemize}
\item \textbf{List} - Ordered and changeable, allows duplicates
\item \textbf{Tuple} - Ordered and unchangeable, allows duplicates
\item \textbf{Set} - Unordered and unindexed, no duplicates
\item \textbf{Dictionary} - Unordered, changeable and indexed, no duplicates
\end{itemize}
\end{frame}

\begin{frame}[fragile]
\frametitle{Lists}
\begin{itemize}
\item Ordered and changeable collection
\item Written with square brackets
\end{itemize}

\begin{lstlisting}
thislist = ["apple", "pineapple", "cherry"]
print(thislist)

# Access items
print(thislist[2])  # cherry

# Change item value
thislist[1] = "carrot"
print(thislist)     # ['apple', 'carrot', 'cherry']

# Check if item exists
if "apple" in thislist:
    print("Yes, 'apple' is in the list")
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{List Methods}
\begin{lstlisting}
thislist = ["apple", "pineapple", "cherry"]

# Length
print(len(thislist))  # 3

# Add items
thislist.append("orange")
thislist.insert(2, "mango")

# Remove items
thislist.remove("cherry")
thislist.pop(1)       # Remove by index
del thislist[0]       # Delete by index

# Copy list
mylist = thislist.copy()
mylist2 = list(thislist)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Tuples}
\begin{itemize}
\item Ordered and unchangeable collection
\item Written with round brackets
\end{itemize}

\begin{lstlisting}
thistuple = ("apple", "pineapple", "cherry")
print(thistuple)

# Access tuple items
print(thistuple[2])  # cherry

# Tuples are unchangeable
# thistuple[1] = "carrot"  # This would cause an error
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Sets}
\begin{itemize}
\item Unordered and unindexed collection
\item Written with curly brackets
\item No duplicate members
\end{itemize}

\begin{lstlisting}
thisset = {"apple", "banana", "cherry"}
print(thisset)

# Cannot access by index
# print(thisset[1])  # This would cause an error

# Loop through set
for x in thisset:
    print(x)

# Check if item exists
print("banana" in thisset)  # True
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Set Methods}
\begin{lstlisting}
thisset = {"apple", "banana", "cherry"}

# Add items
thisset.add("orange")
thisset.update(["mango", "grapes"])

# Remove items
thisset.remove("banana")     # Raises error if not found
thisset.discard("banana")    # No error if not found
thisset.pop()               # Remove random item
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Dictionaries}
\begin{itemize}
\item Unordered, changeable and indexed collection
\item Written with curly brackets with key-value pairs
\end{itemize}

\begin{lstlisting}
thisdict = {
    "brand": "Ford",
    "model": "Mustang",
    "year": 1964
}
print(thisdict)

# Access items
x = thisdict["model"]
x = thisdict.get("model")

# Change values
thisdict["year"] = 2018
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Dictionary Methods}
\begin{lstlisting}
thisdict = {"brand": "Ford", "model": "Mustang", "year": 1964}

# Loop through dictionary
for x in thisdict:
    print(x)  # Print keys

for x in thisdict:
    print(thisdict[x])  # Print values

for x in thisdict.values():
    print(x)  # Print values

for x, y in thisdict.items():
    print(x, y)  # Print key-value pairs
\end{lstlisting}
\end{frame}

\section{Control Structures}

\begin{frame}[fragile]
\frametitle{Selection Statements - If}
\begin{lstlisting}
a = 33
b = 200

if b > a:
    print("b is greater than a")
\end{lstlisting}

\textbf{Important:} Python relies on indentation to define scope.

\begin{lstlisting}
# This will cause an error
if b > a:
print("b is greater than a")  # No indentation
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{If-Elif-Else}
\begin{lstlisting}
a = 33
b = 33

if b > a:
    print("b is greater than a")
elif a == b:
    print("a and b are equal")
else:
    print("a is greater than b")
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Logical Operators in Conditions}
\begin{lstlisting}
# AND operator
if a > b and c > a:
    print("Both conditions are True")

# OR operator
if a > b or a > c:
    print("At least one condition is True")
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{While Loop}
\begin{lstlisting}
i = 1
while i < 6:
    print(i)
    i += 1

# Break statement
i = 1
while i < 6:
    print(i)
    if i == 3:
        break
    i += 1

# Continue statement
i = 0
while i < 6:
    i += 1
    if i == 3:
        continue
    print(i)
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{For Loop}
\begin{lstlisting}
# Loop through list
fruits = ["apple", "banana", "cherry"]
for x in fruits:
    print(x)

# Loop through string
for x in "banana":
    print(x)

# Break in for loop
for x in fruits:
    print(x)
    if x == "banana":
        break
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Range Function}
\begin{lstlisting}
# Basic range
for x in range(6):
    print(x)  # 0, 1, 2, 3, 4, 5

# Range with start parameter
for x in range(2, 6):
    print(x)  # 2, 3, 4, 5

# Range with step parameter
for x in range(2, 30, 3):
    print(x)  # 2, 5, 8, 11, 14, 17, 20, 23, 26, 29
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Nested Loops}
\begin{lstlisting}
adj = ["red", "big", "tasty"]
fruits = ["apple", "banana", "cherry"]

for x in adj:
    for y in fruits:
        print(x, y)
\end{lstlisting}

Output: red apple, red banana, red cherry, big apple, etc.
\end{frame}

\section{File Handling}

\begin{frame}
\frametitle{File Handling Modes}
\begin{center}
\begin{tabular}{|c|l|}
\hline
\textbf{Mode} & \textbf{Description} \\
\hline
"r" & Read (default) \\
"a" & Append \\
"w" & Write \\
"x" & Create \\
\hline
\end{tabular}
\end{center}

Additional modes:
\begin{itemize}
\item "t" - Text mode (default)
\item "b" - Binary mode
\end{itemize}
\end{frame}

\begin{frame}[fragile]
\frametitle{Reading Files}
\begin{lstlisting}
# Open and read file
f = open("demofile.txt", "r")
print(f.read())

# Read specific number of characters
f = open("demofile.txt", "r")
print(f.read(5))

# Read line by line
f = open("demofile.txt", "r")
print(f.readline())

# Always close files
f.close()
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Writing Files}
\begin{lstlisting}
# Append to file
f = open("demofile2.txt", "a")
f.write("Now the file has more content!")
f.close()

# Overwrite file
f = open("demofile3.txt", "w")
f.write("Woops! I have deleted the content!")
f.close()

# Create new file
f = open("myfile.txt", "x")  # or "a" or "w"
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Delete Files}
\begin{lstlisting}
import os

# Remove file
os.remove("demofile.txt")

# Check if file exists before deleting
if os.path.exists("demofile.txt"):
    os.remove("demofile.txt")
else:
    print("The file does not exist")

# Remove folder
os.rmdir("myfolder")
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]
\frametitle{Arrays in Python}
\begin{itemize}
\item Arrays can hold multiple values under single name
\item Access values by index number
\end{itemize}

\begin{lstlisting}
# Instead of:
car1 = "Ford"
car2 = "Volvo"
car3 = "BMW"

# Use array:
cars = ["Ford", "Volvo", "BMW"]

# Access elements
x = cars[0]  # Ford

# Modify elements
cars[0] = "Toyota"
\end{lstlisting}
\end{frame}

\begin{frame}[fragile]

\begin{lstlisting}
cars = ["Ford", "Volvo", "BMW"]

# Length of array
x = len(cars)

# Loop through array
for x in cars:
    print(x)

# Add element
cars.append("Honda")

# Remove elements
cars.pop(1)           # Remove by index
cars.remove("Volvo")  # Remove by value
\end{lstlisting}
\end{frame}
